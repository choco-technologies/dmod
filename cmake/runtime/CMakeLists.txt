# Build only the module-side runtime, using the consumer's compiler and flags.
set(DMOD_GENERATE_MAKEFILES OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_definitions(DMOD_MODULE ON=1 OFF=0)
add_compile_options(-ffunction-sections -fdata-sections -fno-builtin -fno-stack-protector)
configure_file("${DMOD_DIR}/dmod-config.h.in" "${DMOD_BUILD_DIR}/dmod-config.h")

add_library(dmod_inc INTERFACE)
target_include_directories(dmod_inc INTERFACE "${DMOD_INC_DIR}" "${DMOD_BUILD_DIR}")
target_compile_definitions(dmod_inc INTERFACE DMOD_MODULE ON=1 OFF=0)
if(DEFINED DMOD_ARCH)
    target_compile_definitions(dmod_inc INTERFACE DMOD_ARCH="${DMOD_ARCH}")
endif()
if(DEFINED DMOD_CPU)
    target_compile_definitions(dmod_inc INTERFACE DMOD_CPU="${DMOD_CPU}")
endif()
add_subdirectory("${DMOD_SRC_DIR}/common" common)
add_subdirectory("${DMOD_SRC_DIR}/module" module)

add_library(dmod INTERFACE)
target_link_libraries(dmod INTERFACE dmod_inc dmod_common dmod_module)
if(DMOD_USE_FASTLZ)
    add_subdirectory("${DMOD_DIR}/lib/third-party/FastLZ" fastlz)
    target_link_libraries(dmod INTERFACE dmod_fastlz)
endif()
